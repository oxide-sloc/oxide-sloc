# oxide-sloc Jenkins pipeline — required plugins
#
# See ci/jenkins/README.md "Installing plugins" for full instructions.
# Run ci/jenkins/preflight.sh after installing — check (c) verifies all plugins.
#
# ── Path 1: Docker (online) ───────────────────────────────────────────────────
#   docker exec -u root <container> jenkins-plugin-cli \
#     --plugins $(grep -Ev '^#|^$' ci/jenkins/plugins.txt | awk '{print $1}' | tr '\n' ' ')
#
# ── Path 2: Docker (air-gapped) ───────────────────────────────────────────────
# Download each .hpi on a networked machine, then transfer and install offline:
#
#   # Download (on networked machine):
#   while IFS= read -r line; do
#     [[ "$line" =~ ^#|^$ ]] && continue
#     id=$(echo "$line" | awk '{print $1}')
#     curl -L -o "${id}.hpi" "https://updates.jenkins.io/latest/${id}.hpi"
#   done < ci/jenkins/plugins.txt
#
#   # Install (on air-gapped host — repeat for each .hpi):
#   docker exec -u root <container> \
#     cp /host/path/<plugin>.hpi /var/jenkins_home/plugins/<plugin>.hpi
#   # Then restart Jenkins to load the plugins.
#
#   Alternatively: Manage Jenkins → Plugins → Advanced → Deploy Plugin → Upload .hpi
#
# ── Path 3: Native / systemd (Jenkins CLI jar) ────────────────────────────────
#   source ci/jenkins/.env
#   curl -sS -o jenkins-cli.jar "${JENKINS_URL}/jnlpJars/jenkins-cli.jar"
#   java -jar jenkins-cli.jar -s "${JENKINS_URL}" -auth "${JENKINS_USER}:${JENKINS_TOKEN}" \
#       install-plugin $(grep -Ev '^#|^$' ci/jenkins/plugins.txt | awk '{print $1}')
#   java -jar jenkins-cli.jar -s "${JENKINS_URL}" -auth "${JENKINS_USER}:${JENKINS_TOKEN}" safe-restart

# ── Core pipeline support ──────────────────────────────────────────────────────
workflow-aggregator         # Declarative Pipeline syntax (stages, when, post)
pipeline-utility-steps      # readJSON, writeJSON helpers used in post { success }
ws-cleanup                  # cleanWs() in post { cleanup }

# ── Source control ─────────────────────────────────────────────────────────────
git                         # GitSCM checkout step

# ── Credentials ───────────────────────────────────────────────────────────────
credentials-binding         # credentials() binding in environment blocks

# ── Reporting — required for full pipeline feature set ────────────────────────
htmlpublisher               # publishHTML() — "SLOC Report" browsable HTML report sidebar link
plot                        # plot() — SLOC trend charts across builds
junit                       # junit() — "Test Result" sidebar link (cargo-nextest runs)
coverage                    # recordCoverage() — native LCOV/Cobertura integration; shows
                            #   line %, branch %, function % on each build page, per-file
                            #   drill-down, build-over-build trend chart, and quality-gate
                            #   enforcement — all without manual CSV or shell parsing.
                            #   Supersedes the manual lcov.info threshold check.

# ── UX quality-of-life ─────────────────────────────────────────────────────────
pipeline-stage-view         # Stage visualization in the job UI (Blue Ocean lite)
timestamper                 # Timestamps on every console output line (used by options{ timestamps() })
ansicolor                   # ANSI colour in Rust output (optional — the ansiColor() pipeline
                            #   directive was removed so this is NOT required; install only if you
                            #   re-add ansiColor('xterm') to the Jenkinsfile options block)

# ── Run-page enrichment (optional — pipeline degrades to native dashboard) ─────
# These add richer visuals to the Jenkins run/job page itself. They are handled
# by detect-capabilities.py + install-plugins.sh: a system-admin build installs
# them (offline or via the update centre); otherwise the report shows a warning
# banner and the self-contained native dashboard. Never required.
badge                       # addBadge()/addShortText() — highlight SLOC/test/coverage on the run row (optional)
warnings-ng                 # recordIssues() — trend graphs + per-issue drill-down for style findings (optional)

# ── Job automation ─────────────────────────────────────────────────────────────
job-dsl                     # Executes ci/jenkins/seed-job.groovy to create the pipeline job
copyartifact                # Copies artifacts to/from downstream jobs in Pipeline-of-Pipelines

# ── Triggers (optional) ────────────────────────────────────────────────────────
# Enables the release/tag webhook trigger rendered by render-job-config.sh when
# SLOC_ENABLE_WEBHOOK_TRIGGER=1 (a GitHub/Bitbucket release or tag push starts the
# scan and maps the pushed tag into SCAN_REF). Optional — only needed for webhook
# triggering; the pipeline runs fine without it. See docs/ci-integrations.md.
generic-webhook-trigger          # GenericTrigger — release/tag webhook -> pipeline (optional)

# ── SCM integrations (optional) ───────────────────────────────────────────────
# These two plugins are genuinely optional — the pipeline degrades gracefully
# when they are absent (the post{always} bitbucketStatusNotify call is wrapped
# in try/catch).  preflight.sh prints [warn] instead of [fail] for lines whose
# comment contains "(optional)", so a missing Bitbucket plugin will NOT block
# the bootstrap.  Install them only when using Bitbucket as the SCM source.
bitbucket                        # Bitbucket Branch Source (optional — Bitbucket SCM only)
bitbucket-build-status-notifier  # Posts commit statuses to Bitbucket (optional — Bitbucket SCM only)
